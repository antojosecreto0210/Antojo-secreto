    <script>
        document.getElementById('productForm').addEventListener('submit', function(e) {
            e.preventDefault();
            const title = document.getElementById('title').value;
            const category = document.getElementById('category').value;
            const price = document.getElementById('price').value;
            const stock = document.getElementById('stock').value;
            const desc = document.getElementById('desc').value;
            const imgNameInput = document.getElementById('imgName').value.trim();
            const imgFile = document.getElementById('imgFile').files[0];

            function saveAndRedirect(imgSrc) {
                let products = JSON.parse(localStorage.getItem('antojo_products')) || [];
                const newProd = {
                    id: Date.now(),
                    title,
                    category,
                    price,
                    stock,
                    desc,
                    img: imgSrc || 'https://via.placeholder.com/300x220?text=Antojo+Secreto'
                };
                products.push(newProd);
                localStorage.setItem('antojo_products', JSON.stringify(products));
                alert('¡Producto guardado con éxito!');
                window.location.href = 'index.html';
            }

            if (imgFile) {
                const reader = new FileReader();
                reader.onload = function(uploadEvent) {
                    saveAndRedirect(uploadEvent.target.result);
                };
                reader.readAsDataURL(imgFile);
            } else {
                saveAndRedirect(imgNameInput || 'https://via.placeholder.com/300x220?text=Antojo+Secreto');
            }
        });
    </script>
